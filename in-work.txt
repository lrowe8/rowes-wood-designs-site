project-root/
├── src/
│   ├── config/          # Environment variables and global settings
│   ├── controllers/     # Request/Response handlers (Transport layer)
│   ├── middleware/      # Auth, logging, and error handling
│   ├── models/          # Database schemas (e.g., Mongoose or Prisma)
│   ├── routes/          # API endpoint definitions
│   ├── services/        # Core business logic (Reusable tasks)
│   ├── types/           # Global TypeScript interfaces and declarations
│   ├── utils/           # Reusable helper functions
│   ├── app.ts           # Express app configuration
│   └── server.ts        # Entry point for starting the server
├── tests/               # Unit and integration tests
├── dist/                # Compiled JavaScript files (generated by tsc)
├── .env                 # Secret environment variables
├── package.json         # Dependencies and scripts
└── tsconfig.json        # TypeScript compiler settings
